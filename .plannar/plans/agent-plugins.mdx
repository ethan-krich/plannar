### Editor app changes

All changes are in `apps/editor/src/`. Two new files, one modified file.

**New file: `agent-state.ts`** — detects the calling agent from a `?agent=` query param.

```tsx
export type AgentName = "opencode" | "claude" | "codex" | null;

export type AgentConfig = {
  name: AgentName;
  label: string;
  icon: string; // inline SVG string
};

const AGENTS: Record<string, AgentConfig> = {
  opencode: { name: "opencode", label: "OpenCode", icon: "<OpenCode SVG>" },
  claude: { name: "claude", label: "Claude Code", icon: "<Claude SVG>" },
  codex: { name: "codex", label: "Codex CLI", icon: "<Codex SVG>" },
};

export function useAgent(): AgentConfig | null {
  const params = new URLSearchParams(window.location.search);
  const agent = params.get("agent")?.toLowerCase();
  return agent && AGENTS[agent] ? AGENTS[agent] : null;
}
```

**New file: `agent-button.tsx`** — replaces the "Generate prompt (N) / Copy" button when an agent is active.

```tsx
export function AgentButton({ agent, comments }) {
  const [submitted, setSubmitted] = useState(false);

  const handleSubmit = useCallback(() => {
    const prompt = generatePrompt(comments);
    void navigator.clipboard.writeText(prompt).then(() => {
      setSubmitted(true);
      setTimeout(() => setSubmitted(false), 2000);
      toast(`Prompt copied — switch to ${agent.label} and paste (Cmd+V)`);
    });
  }, [agent, comments]);

  return (
    <button
      onClick={handleSubmit}
      className="fixed bottom-6 right-20 z-40 inline-flex items-center gap-1.5 rounded-lg bg-primary px-3 py-2 text-xs font-medium text-primary-foreground shadow-lg hover:bg-primary/80 transition-colors"
    >
      {submitted ? (
        <Check className="size-3" />
      ) : (
        <span dangerouslySetInnerHTML={{ __html: agent.icon }} className="size-3" />
      )}
      {submitted ? "Submitted" : `Submit to ${agent.label} (${comments.length})`}
    </button>
  );
}
```

```ts
import { definePlugin } from "opencode";

export default definePlugin((ctx) => {
  // /create-plan <name>
  ctx.addCommand("create-plan", async ({ args }) => {
    const planName = args.join("-");
    const cwd = ctx.workspace;

    // 1. Ensure .plannar/ exists
    if (!existsSync(join(cwd, ".plannar"))) {
      await exec("npx plannar@latest init --yes", cwd);
    }

    // 2. Check if editor is running
    const status = await exec("npx plannar@latest status", cwd);
    let port = 5173;
    // ... parse port from status output ...

    // 3. If not running, start editor
    if (status.includes("not running")) {
      exec("npx plannar@latest editor --port " + port, cwd, { detached: true });
    }

    // 4. Load the plannar skill
    ctx.loadSkill("plannar");

    // 5. Inject plan filename into context
    ctx.setEnv("PLANNAR_PLAN_NAME", planName);
    ctx.setEnv("PLANNAR_EDITOR_URL", `http://localhost:${port}`);

    // 6. Attach plan file to session when created
    ctx.onFileCreated(join(cwd, ".plannar", "plans", `${planName}.mdx`), (path) => {
      ctx.readFile(path); // loads into session context
    });

    return `Create the plan. Editor URL: http://localhost:${port}/${planName}?agent=opencode`;
  });

  // /export-plan
  ctx.addCommand("export-plan", async () => {
    const planName = ctx.getEnv("PLANNAR_PLAN_NAME");
    await exec(`npx plannar@latest export ${planName}`, ctx.workspace);
  });

  // Install hook — during plugin install, install the skill too
  ctx.onInstall(async () => {
    await exec("npx skills add ethan-krich/plannar@plannar", cwd);
  });
});
```

### New CLI subcommand: `plannar agent-init`

Registered in `packages/plannar/src/cli.ts`, implemented in `packages/plannar/src/commands/agent-init.ts`.

```ts
export default defineCommand({
  meta: {
    name: "agent-init",
    description: "Scaffold plannar integration for a coding agent",
  },
  args: {
    agent: {
      type: "positional",
      description: "Agent to initialize: opencode | claude | codex",
      required: true,
    },
    global: {
      type: "boolean",
      description: "Install globally instead of project-level",
      default: false,
    },
  },
  async run({ args }) {
    const agent = args.agent as string;
    const isGlobal = args.global as boolean;
    const cwd = isGlobal ? os.homedir() : process.cwd();

    // 1. Scaffold agent-specific config files
    switch (agent) {
      case "opencode":
        scaffoldOpenCode(cwd);
        break;
      case "claude":
        scaffoldClaudeCode(cwd, isGlobal);
        break;
      case "codex":
        scaffoldCodex(cwd, isGlobal);
        break;
      default:
        console.error(`Unknown agent: ${agent}`);
        process.exit(1);
    }

    // 2. Install plannar skill at same level as plugin
    console.log("Installing plannar skill...");
    const scopeFlag = isGlobal ? " -g" : "";
    execSync(`npx skills add ethan-krich/plannar@plannar${scopeFlag}`, {
      cwd: isGlobal ? undefined : cwd,
      stdio: "inherit",
    });

    // 3. Run plannar init if .plannar/ doesn't exist
    const plannarDir = join(cwd, ".plannar");
    if (!existsSync(plannarDir)) {
      console.log("No .plannar/ found — running plannar init...");
      execSync("npx plannar@latest init", {
        cwd,
        stdio: "inherit",
        // init prompts for skill install — but we already did it
        // Consider adding a --skip-skill flag to init
      });
    }

    console.log(`✓ Plannar integration for ${agent} is ready.`);
  },
});
```

## Risk assessment

| Risk                                                                        | Mitigation                                                                                                                                                                                      |
| --------------------------------------------------------------------------- | ----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| OpenCode plugin API differs from assumptions                                | Verify against opencode docs before implementing `ctx.addCommand`, `ctx.onFileCreated`, etc.                                                                                                    |
| Port conflicts if editor already running on different port                  | `plannar status` scans next 10 ports — if busy, use `--port N+1`                                                                                                                                |
| Clipboard API blocked on non-localhost origins                              | Only affects remote `--host` deploys, which are edge case. Toast instructions are clear.                                                                                                        |
| `plannar init --yes` needs stdin-is-TTY bypass                              | Add `--yes` flag. When set, skip interactive prompts and auto-accept all defaults including skill install.                                                                                      |
| Brand icon licensing                                                        | Source from each agent's official brand kits; store as inline SVG data URIs.                                                                                                                    |
| Skill install prompt from `init` conflicts with `agent-init` pre-installing | When `agent-init` runs before `init`, the `init` skill prompt should be non-interactive if `--yes` is passed. Create a `--skip-skill` flag on init for cases where skill was already installed. |
